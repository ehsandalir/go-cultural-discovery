import {useEffect,useRef,useState} from 'react';
import {useQuery,useQueryClient} from '@tanstack/react-query';
import {getBackupOverview,saveBackupConfig,createProjectBackup,continueProjectBackup,downloadProjectBackup,retryProjectBackupDelivery,startBackupFileUpload,uploadBackupFilePart,inspectBackupFile,restoreBackupFile} from '@/lib/backup.functions';
import './backup-panel.css';
import {tehranInput,fromTehranInput} from '@/lib/backup-time';
export default function BackupPanel({lang}:{lang:'en'|'fa'}){
 const tr=(e:string,p:string)=>lang==='fa'?p:e,cache=useQueryClient(),alive=useRef(true);
 useEffect(()=>{alive.current=true;return()=>{alive.current=false}},[]);
 const q=useQuery({queryKey:['go-backups'],queryFn:()=>getBackupOverview(),retry:false});
 const [password,setPassword]=useState(''),[busy,setBusy]=useState(false),[message,setMessage]=useState(''),[progress,setProgress]=useState(0),[file,setFile]=useState<File|null>(null),[confirmed,setConfirmed]=useState(false),[inspected,setInspected]=useState<Awaited<ReturnType<typeof inspectBackupFile>>|null>(null);
 const request=useRef(crypto.randomUUID());
 const error=(e:unknown)=>String(e).includes('password')?tr('Use the correct backup password (at least 12 characters).','رمز درست بک‌آپ را وارد کن؛ حداقل ۱۲ کاراکتر.'):String(e).includes('safety_snapshot')?tr('Create a fresh backup before restoring data.','پیش از بازیابی، یک بک‌آپ تازه بساز.'):String(e).includes('sourceNeedsDeployment')?tr('Source recovery requires a site deployment.','بازیابی کد نیازمند انتشار سایت است.'):tr('The operation did not complete. Current status is shown below; refresh and retry.','عملیات کامل نشد؛ وضعیت در فهرست ثبت شده است. تازه‌سازی کن و دوباره تلاش کن.');
 async function advance(id:string){while(alive.current){const r=await continueProjectBackup({data:{id}});if(!alive.current)return;setProgress(r.progress);if(r.done)break;await new Promise(r=>setTimeout(r,500));}await q.refetch();}
 async function create(){if(busy)return;setBusy(true);setMessage('');setProgress(0);try{const r=await createProjectBackup({data:{requestId:request.current,password}});await advance(r.id);request.current=crypto.randomUUID();setMessage(tr('Both encrypted packages are ready. Check delivery status below.','دو بستهٔ رمزگذاری‌شده آماده شدند؛ وضعیت ارسال را در فهرست ببین.'));}catch(e){setMessage(error(e));void q.refetch()}finally{if(alive.current)setBusy(false)}}
 async function inspect(){if(!file||busy)return;setBusy(true);setMessage('');setInspected(null);setConfirmed(false);try{const count=Math.ceil(file.size/(3*1024*1024));if(count<1||count>400)throw Error('file_size');const r=await startBackupFileUpload({data:{parts:count}});for(let i=0;i<count;i++){const blob=file.slice(i*3*1024*1024,(i+1)*3*1024*1024);const data=await new Promise<string>((resolve,reject)=>{const reader=new FileReader();reader.onload=()=>resolve(String(reader.result).split(',')[1]!);reader.onerror=reject;reader.readAsDataURL(blob)});await uploadBackupFilePart({data:{id:r.id,index:i,data}});setProgress(Math.floor((i+1)/count*100))}setInspected(await inspectBackupFile({data:{id:r.id,password}}));setMessage(tr('File and checksums verified. Review the recovery scope.','فایل و صحت محتوا تأیید شدند؛ محدودهٔ بازیابی را بررسی کن.'));}catch(e){setMessage(error(e))}finally{setBusy(false)}}
 if(q.isPending)return <p className="gs-backup" role="status">{tr('Loading backups…','در حال دریافت بک‌آپ‌ها…')}</p>;
 if(q.isError)return <p className="gs-backup" role="alert">{tr('Backups unavailable.','بک‌آپ‌ها در دسترس نیستند.')} <button onClick={()=>void q.refetch()}>{tr('Retry','تلاش دوباره')}</button></p>;
 return <main className="gs-backup" dir="ltr"><h2 dir="auto">{tr('Backup & recovery','بک‌آپ و بازیابی')}</h2><p dir="auto">{tr('Two separate encrypted packages: project source, site assets and catalog; user accounts, rooms and personal data. External audio files remain on their original server; GO backs up their links.','دو بستهٔ مستقل و رمزگذاری‌شده: کد، فایل‌های سایت و بانک آثار؛ حساب‌ها، اتاق‌ها و اطلاعات کاربران. فایل صوتی روی سرور خارجی می‌ماند و GO از لینک آن بک‌آپ می‌گیرد.')}</p><label>{tr('Backup password','رمز مستقل بک‌آپ')}<input type="password" autoComplete="new-password" minLength={12} maxLength={300} value={password} onChange={e=>setPassword(e.target.value)} placeholder={tr('12 characters minimum','حداقل ۱۲ کاراکتر')}/></label><button disabled={busy||!q.data.configured||password.length<12&&!q.data.settings?.hasPassword} onClick={()=>void create()}>{tr('Create both packages','ساخت دو بسته')}</button>{busy&&<progress aria-label={tr('Backup progress','پیشرفت عملیات')} max={100} value={progress}/>}<p role="status" dir="auto">{message}</p>
 <section><h3 dir="auto">{tr('History & downloads','سوابق و دانلود')}</h3><button disabled={busy} onClick={()=>void q.refetch()}>{tr('Refresh','تازه‌سازی')}</button>{q.data.runs.map(run=><article key={run.id}><time dir="auto">{new Date(run.created_at).toLocaleString(lang==='fa'?'fa-IR':'en-GB',{timeZone:'Asia/Tehran'})}</time><span dir="auto">{({ready:tr('Ready','آماده'),preparing:tr('Preparing','در حال آماده‌سازی'),running:tr('Running','در حال اجرا'),failed:tr('Failed','ناموفق'),delivery_failed:tr('Ready · delivery failed','آماده · ارسال ناموفق')} as Record<string,string>)[run.status]??run.status}</span>{run.error&&<small>{run.error}</small>}<div>{(['project','users'] as const).map(kind=><button key={kind} disabled={busy||!(kind==='project'?run.project_key:run.users_key)} onClick={async()=>{try{const r=await downloadProjectBackup({data:{id:run.id,kind}});const a=document.createElement('a');a.href=r.url;a.download='';a.click()}catch(e){setMessage(error(e))}}>{kind==='project'?tr('Project package','بستهٔ پروژه'):tr('Users package','بستهٔ کاربران')}</button>)}{['preparing','running'].includes(run.status)&&<button disabled={busy} onClick={async()=>{setBusy(true);try{await advance(run.id)}catch(e){setMessage(error(e))}finally{setBusy(false)}}>{tr('Continue','ادامهٔ آماده‌سازی')}</button>}{run.status==='delivery_failed'&&<button disabled={busy} onClick={async()=>{setBusy(true);try{await retryProjectBackupDelivery({data:{id:run.id}});await q.refetch()}catch(e){setMessage(error(e))}finally{setBusy(false)}}>{tr('Retry delivery','تلاش دوباره برای ارسال')}</button>}</div></article>)}</section>
 <BackupConfig key={q.data.revision} current={q.data} password={password} tr={tr} done={()=>void q.refetch()}/>
 <section><h3 dir="auto">{tr('Upload & recover','آپلود و بازیابی')}</h3><p dir="auto">{tr('Recovery merges saved data without removing newer rows. Create a fresh safety backup first. Source code and static site files need a separate build/deployment; this panel does not redeploy the site.','بازیابی، داده‌های ذخیره‌شده را برمی‌گرداند و ردیف‌های جدیدتر را حذف نمی‌کند. ابتدا یک بک‌آپ ایمنی تازه بساز. کد و فایل‌های ثابت سایت به بازسازی و انتشار جداگانه نیاز دارند؛ این پنل سایت را دوباره منتشر نمی‌کند.')}</p><input type="file" accept=".gobackup" disabled={busy} onChange={e=>{setFile(e.target.files?.[0]??null);setInspected(null);setConfirmed(false)}}/><button disabled={busy||!file||password.length<12} onClick={()=>void inspect()}>{tr('Upload & verify','آپلود و بررسی')}</button>{inspected&&<><p dir="auto">{inspected.kind==='project'?tr('Project','پروژه'):tr('Users','کاربران')} · {inspected.entries} {tr('verified entries','بخش تأییدشده')} · {new Date(inspected.createdAt).toLocaleString()}</p><label><input type="checkbox" checked={confirmed} onChange={e=>setConfirmed(e.target.checked)}/>{tr('I confirm recovery of this package into current GO data.','بازیابی این بسته در اطلاعات فعلی GO را تأیید می‌کنم.')}</label><button disabled={!confirmed||busy} onClick={async()=>{setBusy(true);try{const r=await restoreBackupFile({data:{id:inspected.id,confirmed:true}});setMessage(r.sourceNeedsDeployment?tr('Shared data recovered. Source code still requires deployment.','اطلاعات مشترک بازیابی شد؛ کد سایت هنوز به انتشار نیاز دارد.'):tr('User data recovered. Sign in again if needed.','اطلاعات کاربران بازیابی شد؛ در صورت نیاز دوباره وارد شو.'));setInspected(null);await cache.invalidateQueries()}catch(e){setMessage(error(e))}finally{setBusy(false)}}>{tr('Recover saved data','بازیابی اطلاعات ذخیره‌شده')}</button></>}</section></main>
}
function BackupConfig({current,password,tr,done}:{current:Awaited<ReturnType<typeof getBackupOverview>>;password:string;tr:(e:string,p:string)=>string;done:()=>void}){
 const original=current.settings;
 const [enabled,setEnabled]=useState(original?.enabled??false),[hours,setHours]=useState(original?.intervalHours??24),[first,setFirst]=useState(tehranInput(original?.firstRunAt??Date.now()+3600000)),[destination,setDestination]=useState(original?.destination??''),[token,setToken]=useState(''),[email,setEmail]=useState(original?.email??''),[from,setFrom]=useState(original?.emailFrom??''),[key,setKey]=useState(''),[busy,setBusy]=useState(false),[message,setMessage]=useState('');
 return <details><summary>{tr('Automatic schedule & delivery','برنامهٔ خودکار و ارسال')}</summary><p dir="auto">{tr('Scheduled jobs run on the server while GO is closed. Destination must accept HTTPS PUT (WebDAV / upload endpoint). Email sends two encrypted download links using your configured Resend sender and API key. Recipients need access to GO.','برنامه روی سرور اجرا می‌شود، حتی وقتی GO بسته است. مقصد باید HTTPS PUT را بپذیرد؛ مانند WebDAV یا نشانی آپلود. ایمیل شامل دو لینک دانلود رمزگذاری‌شده است و از فرستنده و کلید Resend شما استفاده می‌کند. گیرنده باید به GO دسترسی داشته باشد.')}</p><label><input type="checkbox" checked={enabled} onChange={e=>setEnabled(e.target.checked)}/>{tr('Enable automatic backups','فعال‌کردن بک‌آپ خودکار')}</label><label>{tr('Every … hours','هر … ساعت')}<input type="number" min={1} max={8760} value={hours} onChange={e=>setHours(Number(e.target.value))}/></label><label>{tr('First run · Tehran time','اولین اجرا · زمان تهران')}<input type="datetime-local" value={first} onChange={e=>setFirst(e.target.value)}/></label><label>{tr('Destination folder URL','آدرس پوشهٔ مقصد')}<input type="url" value={destination} onChange={e=>setDestination(e.target.value)} placeholder="https://…"/></label><label>{tr('Destination access token (if required)','توکن دسترسی مقصد، در صورت نیاز')}<input type="password" autoComplete="off" value={token} onChange={e=>setToken(e.target.value)} placeholder={original?.hasDestinationToken?tr('Saved; leave blank to keep','ذخیره شده؛ برای حفظ خالی بگذار'):''}/></label><label>{tr('Backup email','ایمیل بک‌آپ')}<input type="email" value={email} onChange={e=>setEmail(e.target.value)}/></label><label>{tr('Verified sender','فرستندهٔ تأییدشده')}<input value={from} onChange={e=>setFrom(e.target.value)}/></label><label>Resend API key<input type="password" autoComplete="off" value={key} onChange={e=>setKey(e.target.value)} placeholder={original?.hasEmailKey?tr('Saved; leave blank to keep','ذخیره شده؛ برای حفظ خالی بگذار'):''}/></label><button disabled={busy} onClick={async()=>{setBusy(true);setMessage('');try{await saveBackupConfig({data:{revision:current.revision,settings:{enabled,intervalHours:hours,firstRunAt:fromTehranInput(first),destination,destinationToken:token,email,emailFrom:from,emailApiKey:key,password}}});done();setMessage(tr('Schedule and delivery settings saved.','برنامه و تنظیمات ارسال ذخیره شدند.'))}catch{setMessage(tr('Not saved. Check the password, time and destination.','ذخیره نشد؛ رمز، زمان و مقصد را بررسی کن.'))}finally{setBusy(false)}}>{tr('Save schedule & delivery','ذخیرهٔ برنامه و ارسال')}</button><p role="status" dir="auto">{message}</p></details>
}
